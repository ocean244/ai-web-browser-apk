 442      private fun navigateTo(rawInput: String) {
 443          val trimmed = rawInput.trim()
 444          if (trimmed.isEmpty()) return
 445          val normalized = if (
 446              trimmed.startsWith("http://", ignoreCase = true) ||
 447              trimmed.startsWith("https://", ignoreCase = true) ||
 448              trimmed.startsWith("file://", ignoreCase = true) ||
 449              trimmed.startsWith("about:", ignoreCase = true)
 450          ) {
 451              trimmed
 452          } else if (trimmed.contains(".") && !trimmed.contains(" ")) {
 453              "https://$trimmed"
 454          } else {
 455              "https://duckduckgo.com/?q=${Uri.encode(trimmed)}"
 456          }
 457          urlInput.setText(normalized)
 458          webView.loadUrl(normalized)
 459      }

 466              if (cleanUrl.isEmpty()) return
 467              mainHandler.post {

 512          @JavascriptInterface
 513          fun postChatCompletion(
 514              endpoint: String?,
 515              model: String?,
 516              apiKey: String?,
 517              payloadJson: String?
 518          ): String {
 519              val targetUrl = endpoint?.trim().orEmpty()
 520              if (targetUrl.isEmpty()) {
 521                  return JSONObject().put("error", "Brak endpointu AI").toString()
 522              }
 523              val rawBody = payloadJson?.trim().orEmpty()
 524              if (rawBody.isEmpty()) {
 525                  return JSONObject().put("error", "Pusty ładunek JSON").toString()
 526              }
 527  
 528              val normalizedBody = try {
 529                  val obj = JSONObject(rawBody)
 530                  if (!model.isNullOrBlank()) {
 531                      obj.put("model", model.trim())
 532                  }
 533                  obj.toString()
 534              } catch (e: Exception) {
 535                  return JSONObject().put("error", "Niepoprawny JSON żądania: ${e.message}").toString()
 536              }
 537  
 538              return try {
 539                  val mediaType = "application/json; charset=utf-8".toMediaType()
 540                  val body = normalizedBody.toRequestBody(mediaType)
 541                  val reqBuilder = Request.Builder()
 542                      .url(targetUrl)
 543                      .post(body)
 544                      .header("Content-Type", "application/json")
 545                      .header("Accept", "application/json")
 546  
 547                  val cleanKey = apiKey?.trim().orEmpty()
 548                  if (cleanKey.isNotEmpty()) {
 549                      reqBuilder.header("Authorization", "Bearer $cleanKey")
 550                  }
 551  
 552                  httpClient.newCall(reqBuilder.build()).execute().use { response ->
 553                      val responseBody = response.body?.string().orEmpty()
 554                      if (!response.isSuccessful) {
 555                          JSONObject()
 556                              .put("error", "HTTP ${response.code}: ${response.message} ${responseBody.take(400)}")
 557                              .toString()
 558                      } else if (responseBody.isBlank()) {
 559                          JSONObject().put("error", "Pusta odpowiedź z endpointu AI").toString()
 560                      } else {
 561                          responseBody
 562                      }
 563                  }
 564              } catch (e: Exception) {
 565                  JSONObject()
 566                      .put("error", "${e.javaClass.simpleName}: ${e.message ?: "błąd połączenia"}")
 567                      .toString()
 568              }
